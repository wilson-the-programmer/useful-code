search.s


--------

.global _start
.text

_start:
    mov x0, #1
    adr x1, data
    mov x2, #12

    mov x3, #0
    mov w4, #65


search:
    ldrb w5, [x1, x3]
    cmp w5, w4
    b.eq found
    add x3, x3, #1
    cmp x3, x2
    b.lo search
    b exit


next_search:
    add x3, x3, #1
    cmp x3, x2
    b.lo search
    b exit

exit:
    mov x0, #0
    mov x8, #93
    svc #0


found:
    mov x0, #1
    adr x1, found_message
    mov x2, #6
    mov x8, #64
    svc #0

    mov x0, x3
    add x0, x0, #48
    strb w0, [sp, #-1]!
    mov x0, #1
    mov x1, sp
    mov x2, #1
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

    b next_search

print:
    mov x8, #64
    svc #0

    mov x0, #0
    mov x8, #93
    svc #0

.data
data:
    .ascii "Aello, ARM!"

found_message:
    .ascii "Found\n"

not_found:
    .ascii "Not found\n"
